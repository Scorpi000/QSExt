# 关键设计决策

## 1. 动态表单生成

### 问题

QuantStudio 使用 `QSArgs` 的 traits 系统定义参数（`Int`、`Str`、`Enum`、`Float` 等），现有 Notebook GUI 的 `ArgSetupDlg` 已经实现了 `QSArgs → ipywidgets` 的映射。Web 版需要复用这个机制。

### 方案

**后端**：将 QSArgs traits 转换为 JSON Schema

```python
from QuantStudio.Core import __QS_Args__
from typing import get_type_hints

def qs_args_to_json_schema(args_class: type[__QS_Args__]) -> dict:
    """将 QSArgs 类转换为 JSON Schema"""
    schema = {
        "type": "object",
        "properties": {},
        "required": []
    }

    for name, trait in args_class.__dict__.items():
        if not name.startswith('_'):
            prop = {}

            if isinstance(trait, Int):
                prop = {"type": "integer", "title": name}
                if trait.min_value is not None:
                    prop["minimum"] = trait.min_value
                if trait.max_value is not None:
                    prop["maximum"] = trait.max_value

            elif isinstance(trait, Float):
                prop = {"type": "number", "title": name}

            elif isinstance(trait, Str):
                prop = {"type": "string", "title": name}

            elif isinstance(trait, Bool):
                prop = {"type": "boolean", "title": name}

            elif isinstance(trait, Enum):
                prop = {
                    "type": "string",
                    "title": name,
                    "enum": trait.options
                }

            elif isinstance(trait, List):
                if isinstance(trait.item_trait, Enum):
                    prop = {
                        "type": "array",
                        "title": name,
                        "items": {"type": "string", "enum": trait.item_trait.options}
                    }

            elif isinstance(trait, File):
                prop = {"type": "string", "title": name, "format": "data-url"}

            elif isinstance(trait, Directory):
                prop = {"type": "string", "title": name, "format": "data-url"}

            if prop:
                schema["properties"][name] = prop
                if trait.required:
                    schema["required"].append(name)

    return schema


# API 端点
@app.get("/api/args/{class_name}/schema")
async def get_arg_schema(class_name: str):
    """获取 QSArgs 类的 JSON Schema"""
    args_class = get_args_class(class_name)
    return qs_args_to_json_schema(args_class)
```

## 2. 长时间任务处理

### 问题

回测、风险模型计算可能耗时数秒到数分钟，同步请求会超时。

### 方案

**异步任务 + WebSocket 进度推送**

```python
# 后端：任务管理器
import asyncio
import uuid
from typing import Dict, Any
from fastapi import WebSocket

class TaskManager:
    """异步任务管理器"""

    def __init__(self):
        self._tasks: Dict[str, asyncio.Task] = {}
        self._results: Dict[str, Any] = {}
        self._progress: Dict[str, float] = {}
        self._websockets: Dict[str, list[WebSocket]] = {}

    async def submit(
        self,
        task_id: str,
        coro,
        websocket: WebSocket = None
    ):
        """提交异步任务"""
        if websocket:
            self._websockets.setdefault(task_id, []).append(websocket)

        task = asyncio.create_task(
            self._execute(task_id, coro)
        )
        self._tasks[task_id] = task

    async def _execute(self, task_id: str, coro):
        """执行任务并更新进度"""
        try:
            # 执行协程，传入进度回调
            result = await coro(
                progress_callback=lambda p: self._update_progress(task_id, p)
            )
            self._results[task_id] = result
            await self._notify(task_id, {"status": "completed", "progress": 1.0})
        except Exception as e:
            await self._notify(task_id, {"status": "failed", "error": str(e)})

    async def _update_progress(self, task_id: str, progress: float):
        """更新进度"""
        self._progress[task_id] = progress
        await self._notify(task_id, {"status": "running", "progress": progress})

    async def _notify(self, task_id: str, data: dict):
        """通知 WebSocket 客户端"""
        for ws in self._websockets.get(task_id, []):
            try:
                await ws.send_json(data)
            except:
                pass

    def get_result(self, task_id: str) -> Any:
        """获取任务结果"""
        return self._results.get(task_id)

    def get_progress(self, task_id: str) -> float:
        """获取任务进度"""
        return self._progress.get(task_id, 0)


# 全局任务管理器
task_manager = TaskManager()


# API 端点
@app.post("/api/backtest/run")
async def run_backtest(config: BacktestConfig):
    """提交回测任务"""
    task_id = str(uuid.uuid4())

    async def execute(progress_callback):
        # 调用 QuantStudio 回测
        return await run_qs_backtest(config, progress_callback)

    await task_manager.submit(task_id, execute)
    return {"task_id": task_id}


@app.get("/api/backtest/tasks/{task_id}/status")
async def get_task_status(task_id: str):
    """获取任务状态"""
    return {
        "task_id": task_id,
        "progress": task_manager.get_progress(task_id),
        "result": task_manager.get_result(task_id)
    }


@app.websocket("/ws/tasks/{task_id}")
async def task_websocket(websocket: WebSocket, task_id: str):
    """WebSocket 连接，接收实时进度"""
    await websocket.accept()
    await task_manager.submit(task_id, None, websocket)

    try:
        while True:
            # 保持连接
            await websocket.receive_text()
    except:
        pass
```

**前端**：WebSocket 连接

```tsx
import { useEffect, useState } from 'react';

interface TaskProgress {
  status: 'pending' | 'running' | 'completed' | 'failed';
  progress: number;
  error?: string;
}

const useTaskProgress = (taskId: string) => {
  const [progress, setProgress] = useState<TaskProgress>({
    status: 'pending',
    progress: 0
  });

  useEffect(() => {
    if (!taskId) return;

    const ws = new WebSocket(`ws://localhost:8000/ws/tasks/${taskId}`);

    ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      setProgress(data);
    };

    ws.onerror = () => {
      setProgress(prev => ({ ...prev, status: 'failed', error: '连接失败' }));
    };

    return () => ws.close();
  }, [taskId]);

  return progress;
};

// 使用示例
const BacktestRunner: React.FC = () => {
  const [taskId, setTaskId] = useState<string | null>(null);
  const progress = useTaskProgress(taskId);

  const handleRun = async (config: BacktestConfig) => {
    const { task_id } = await api.post('/api/backtest/run', config);
    setTaskId(task_id);
  };

  return (
    <div>
      <Button onClick={handleRun}>运行回测</Button>
      {taskId && (
        <Progress
          percent={Math.round(progress.progress * 100)}
          status={progress.status === 'failed' ? 'exception' : 'active'}
        />
      )}
    </div>
  );
};
```

## 3. 数据预览性能

### 问题

因子数据可能包含数百万行，直接加载到前端会导致卡顿。

### 方案

**后端分页 + 前端虚拟滚动**

```python
# 后端：分页查询
@app.get("/api/factor/{table_name}/{factor_name}/data")
async def get_factor_data(
    table_name: str,
    factor_name: str,
    start_date: date,
    end_date: date,
    page: int = Query(1, ge=1),
    page_size: int = Query(100, ge=10, le=1000)
):
    """获取因子数据（分页）"""
    # 获取元数据
    ft = fdb.getTable(table_name)
    dts = ft.getDateTime(factor_name, None, start_date, end_date)
    ids = ft.getID(factor_name, dts[-1] if dts else None)

    # 计算分页
    total = len(dts)
    start_idx = (page - 1) * page_size
    end_idx = min(start_idx + page_size, total)

    # 只读取当前页的数据
    page_dts = dts[start_idx:end_idx]
    data = ft.readData([factor_name], ids, page_dts)

    return {
        "data": data.to_dict(),
        "pagination": {
            "page": page,
            "page_size": page_size,
            "total": total,
            "total_pages": (total + page_size - 1) // page_size
        }
    }
```

## 4. DAG 可视化

### 问题

因子依赖关系是 DAG（有向无环图），需要可视化展示。现有 Notebook GUI 使用 ipycytoscape，但功能有限。

### 自动布局

使用 dagrejs 自动计算节点位置：

```python
# 后端：自动布局
import networkx as nx
from collections import defaultdict

def calculate_dag_layout(nodes: list, edges: list) -> list:
    """使用 dagre 算法计算 DAG 布局"""
    G = nx.DiGraph()

    for node in nodes:
        G.add_node(node["id"])

    for edge in edges:
        G.add_edge(edge["source"], edge["target"])

    # 拓扑排序，计算层级
    levels = {}
    for node in nx.topological_sort(G):
        if G.in_degree(node) == 0:
            levels[node] = 0
        else:
            levels[node] = max(levels[pred] for pred in G.predecessors(node)) + 1

    # 计算位置
    level_nodes = defaultdict(list)
    for node, level in levels.items():
        level_nodes[level].append(node)

    positions = {}
    for level, nodes_in_level in level_nodes.items():
        for i, node in enumerate(nodes_in_level):
            positions[node] = {
                "x": level * 200,
                "y": i * 100
            }

    # 更新节点位置
    for node in nodes:
        node["position"] = positions[node["id"]]

    return nodes
```

### 方案

```tsx
  connections: Connection[];
  activeConnection: string | null;
  setActiveConnection: (id: string) => void;
```

**后端**：使用 Redis 缓存

```python
import redis
import json

redis_client = redis.Redis(host='localhost', port=6379, db=0)

class CacheService:
    """缓存服务"""

    @staticmethod
    async def get(key: str) -> Any:
        """获取缓存"""
        data = redis_client.get(key)
        return json.loads(data) if data else None

    @staticmethod
    async def set(key: str, value: Any, ttl: int = 3600):
        """设置缓存"""
        redis_client.setex(key, ttl, json.dumps(value))

    @staticmethod
    async def delete(key: str):
        """删除缓存"""
        redis_client.delete(key)

    @staticmethod
    async def get_factor_info(qsid: str) -> dict:
        """获取因子信息（带缓存）"""
        cache_key = f"factor:info:{qsid}"
        cached = await CacheService.get(cache_key)
        if cached:
            return cached

        info = qs_registry.get_factor_info(qsid)
        await CacheService.set(cache_key, info, ttl=3600)
        return info
```

## 6. 错误处理

### 统一错误响应格式

```python
from fastapi import HTTPException
from fastapi.responses import JSONResponse

class APIError(HTTPException):
    """API 错误基类"""
    def __init__(self, status_code: int, code: str, message: str, detail: Any = None):
        super().__init__(status_code=status_code, detail={
            "code": code,
            "message": message,
            "detail": detail
        })

class NotFoundError(APIError):
    def __init__(self, resource: str, id: str):
        super().__init__(404, "NOT_FOUND", f"{resource} not found: {id}")

class ValidationError(APIError):
    def __init__(self, message: str, detail: Any = None):
        super().__init__(400, "VALIDATION_ERROR", message, detail)


# 全局异常处理
@app.exception_handler(APIError)
async def api_error_handler(request, exc):
    return JSONResponse(
        status_code=exc.status_code,
        content=exc.detail
    )
```
