## 基本流程

```python
from QSExt.QSRegistry.api import QSGraphDB
from QuantStudio.Factor.api import HDF5DB, fo

# 连接图数据库
gdb = QSGraphDB(args={"IPAddr": "127.0.0.1", "Port": 7687})
gdb.connect()

# 注册因子库
hdb = HDF5DB(args={"MainDir": "/path/to/hdf5"})
hdb.connect()
gdb.registerFactorDB(hdb)

# 存储因子表
ft = hdb.getTable("StockDB")
gdb.storeFactorTable(ft, fdb_name=hdb.Name)

# 构建因子计算链
close = ft.getFactor("Close")
log_close = fo.log()(close)           # PointOperation
lag5 = fo.lag(5)(log_close)           # TimeOperation

# 存储到图数据库（递归存储整条链）
gdb.storeFactors([lag5], tags={lag5.QSID: ["momentum", "price"]})[0]
```

## 检索与重建

```python
# 搜索因子
results = gdb.searchFactors(name="log", operator_type="Point")
for r in results:
    print(r["Name"], r["QSID"])

# 查看依赖图
graph = gdb.getDependencyGraph(lag5.QSID, direction="down")
# 返回: {"root": "...", "nodes": [lag5, log_close, close], "edges": [...]}

# 重建因子
reconstructed = gdb.reconstructFactor(lag5.QSID)
# 可直接在计算引擎中使用: engine.compute([reconstructed])

# 影响分析
impacted = gdb.impactAnalysis(close.QSID)
# 返回所有依赖 Close 价格的因子，按深度排序
```

## 向量语义检索

```python
from QSExt.QSRegistry.api import QSGraphDB

# 连接时配置嵌入模型
gdb = QSGraphDB(args={
    "IPAddr": "127.0.0.1",
    "Port": 7687,
    "EmbeddingModel": "bge-m3",
    "EmbeddingDim": 1024,
})
gdb.connect()

# 存储因子时自动生成嵌入向量
gdb.storeFactors([lag5], tags={lag5.QSID: ["momentum", "price"]})[0]

# 自然语言搜索
results = gdb.searchFactorsByDescription("动量因子", limit=10)
for r in results:
    print(f"[{r['Similarity']:.4f}] {r['Name']} ({r['FactorClass']})")

# 带阈值过滤
results = gdb.searchFactorsByDescription("财务质量", min_score=0.6)
```

## 自定义算子

```python
from QuantStudio.Factor.api import makeFactorOperator

@makeFactorOperator(OperatorType="Point", Name="custom_normalize", Arity=1, DataType="double")
def normalize(f, idt, iid, x, args):
    return (x[0] - np.nanmean(x[0])) / np.nanstd(x[0])

norm_factor = normalize()(close)
gdb.storeFactors([norm_factor], tags={norm_factor.QSID: ["normalization"]})[0]
```

## 回测注册与检索

```python
from QSExt.QSRegistry.api import QSGraphDB
from QuantStudio.BackTest.SectionFactor.api import IC, QuantilePortfolio
from QuantStudio.BackTest.BackTestModel import BTReport

gdb = QSGraphDB()
gdb.connect()

# --- 构建回测计算图 ---
# 假设已有因子: momentum, return_1d
ic_factor = CalcIC(corr_method="spearman")(momentum, return_1d)
ic_node = IC(ic_factor, args={"RollingAvgPeriod": 12, "GenReport": True})

# --- 注册回测到图数据库 ---
bt_qsid = gdb.storeBacktest(ic_node, tags=["A股", "IC分析"])
# QSID 由 BTNode 的内容哈希自动确定，相同配置幂等

# --- 运行回测并存储结果 ---
from QuantStudio.Core.CalcEngine import Engine
engine = Engine()
context = FactorContext(DTRuler=dts, SectionIDs=ids)
output = engine.run([ic_node], context, ...)[0]  # IC.backward_compute 返回的 dict
# output = {"IC": DataFrame, "IC的移动平均": DataFrame,
#           "截面宽度": DataFrame, "统计数据": DataFrame, "Report": "<html>..."}

result_ids = gdb.storeBacktestResults(bt_qsid, output, dtrange=(dts[0], dts[-1]))

# --- 检索回测 ---
# 查找所有使用了 momentum 因子的回测
bts = gdb.searchBacktests(factor_qsid=momentum.QSID)

# 查找所有 IC 类回测
ic_backtests = gdb.searchBacktests(category="SectionFactor", name="IC")

# 获取回测结果摘要
results = gdb.getBacktestResults(bt_qsid)
for r in results:
    print(f"{r['Key']}: {r['DataType']}")  # IC: DataFrame, 统计数据: DataFrame, ...

# --- BTReport 聚合模式 ---
# 多个回测打包为一个报告节点
report = BTReport([ic_node, quantile_node])
gdb.storeBacktest(report, tags=["综合报告"])
# BTReport 自动建立到子 BTNode 的 [:依赖] 关系

# 查看回测 → 因子的反向追溯
factor_backtests = gdb.getFactorBacktests(momentum.QSID)
# 返回所有使用过 momentum 因子的回测列表

# --- 报告注册 ---
# 方式1：直接注册已生成的报告文件
report_id = gdb.storeReport(
    report_path="D:/reports/momentum_report.html",
    factor_qsids=[momentum.QSID],
    bt_qsid=bt_qsid,
    scenario_name="single_factor",
    name="动量因子 — 单因子测试报告"
)

# 方式2：配合 ReportGenerator 一键注册
from QSExt.ReportGenerator.core import register_reports_to_db
register_reports_to_db(
    result=scenario_result,
    factor_qsids=[momentum.QSID],
    output_dir="D:/reports/",
    bt_qsid=bt_qsid,
    scenario_name="single_factor"
)

# --- 检索报告 ---
# 查找某因子的所有报告
reports = gdb.getFactorReports(momentum.QSID)
for r in reports:
    print(f"{r['Name']} -> {r['FilePath']} ({r['FileSize']}B)")

# 搜索报告
reports = gdb.searchReports(scenario_name="single_factor", fmt="html")
```

## 风险库与风险表注册

```python
from QSExt.QSRegistry.api import QSGraphDB
from QuantStudio.Risk.api import HDF5RiskDB

gdb = QSGraphDB()
gdb.connect()

# 注册风险库
risk_db = HDF5RiskDB(args={"MainDir": "/path/to/risk_data"})
risk_db.connect()
gdb.registerRiskDB(risk_db)

# 注册风险表
cov_rt = risk_db.getTable("CovMatrix")
gdb.storeRiskTable(cov_rt, risk_db_name=risk_db.Name)

# 将回测（风险模型）与风险表关联
gdb.linkBacktestRiskTable(bt_qsid=risk_model_qsid, rt_qsid=cov_rt.QSID)

# 检索风险表
tables = gdb.searchRiskTables(name="Cov")
for t in tables:
    print(t["Name"], t["QSID"])
```

## 因子存储器注册

```python
from QSExt.QSRegistry.api import QSGraphDB
from QuantStudio.Factor.HDF5DB import HDF5DB
from QuantStudio.Factor.FactorStorer import FactorStorer

gdb = QSGraphDB()
gdb.connect()

# --- 准备目标因子库和因子 ---
hdb = HDF5DB(args={"MainDir": "/path/to/hdf5"})
hdb.connect()
gdb.registerFactorDB(hdb)

ft = hdb.getTable("TestTable")
gdb.storeFactorTable(ft, fdb_name=hdb.Name)

# --- 构建因子计算链 ---
close = ft.getFactor("Close")
log_close = fo.log()(close)
gdb.storeFactors([log_close], tags={log_close.QSID: ["price"]})

# --- 创建 FactorStorer ---
storer = FactorStorer(
    deps=[log_close],
    args={
        "TargetFDB": hdb,
        "TargetTable": "TestTable",
        "IfExists": "update",
    }
)

# 注册到图数据库
storer_qsid = gdb.storeFactorStorer(storer, tags=["持久化"])
# 自动建立依赖因子、目标因子库、目标因子表的关系

# --- 检索因子存储器 ---
# 查找写入同一目标表的所有存储器
storers = gdb.searchFactorStorers(target_table="TestTable")
for s in storers:
    print(f"{s['Name']}: {s['TargetFDBName']}/{s['TargetTable']}")

# 查找依赖某个因子的所有存储器
storers = gdb.searchFactorStorers(factor_qsid=log_close.QSID)

# 按 QSID 查询
info = gdb.getFactorStorerByQSID(storer_qsid)
print(info["TargetFDBName"], info["TargetTable"], info["IfExists"])
```
